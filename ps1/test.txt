#Hash of concatenation
user@host:~/Uni/OS/ps1$ gcc -Wall -o kit kit.c
user@host:~/Uni/OS/ps1$ cat kit foo.png | sha1sum
3371193542a8be6ff9f8b18f5f78f29bb26da7a5  -
user@host:~/Uni/OS/ps1$ ./kit kit foo.png | sha1sum
3371193542a8be6ff9f8b18f5f78f29bb26da7a5  -
user@host:~/Uni/OS/ps1$ ./kit -o out.bin kit foo.png
user@host:~/Uni/OS/ps1$ sha1sum out.bin
3371193542a8be6ff9f8b18f5f78f29bb26da7a5  out.bin
